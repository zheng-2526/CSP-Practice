cmake_minimum_required(VERSION 3.21)
project(QtTextEditor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(EditorCore STATIC mainwindow.cpp mainwindow.h mainwindow.ui
    documenteditor.cpp documenteditor.h rtfcodec.cpp rtfcodec.h)
target_link_libraries(EditorCore PUBLIC Qt6::Widgets)
target_include_directories(EditorCore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(QtTextEditor main.cpp)
target_link_libraries(QtTextEditor PRIVATE EditorCore)
set_target_properties(QtTextEditor PROPERTIES WIN32_EXECUTABLE ON)
enable_testing()
add_executable(EditorTest tests/editor_test.cpp)
target_link_libraries(EditorTest PRIVATE EditorCore)
add_test(NAME EditorTest COMMAND EditorTest)
